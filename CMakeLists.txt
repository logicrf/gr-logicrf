cmake_minimum_required(VERSION 3.16)
project(gr-logicrf VERSION 1.2.2 LANGUAGES C CXX)
# Native IQ hot paths need optimized code; explicit Debug and multi-config builds remain available.

  message(FATAL_ERROR "H102 TX v3 protocol source pin mismatch")

target_compile_options(gnuradio-logicrf PRIVATE -Wall -Wextra -Wpedantic)
# Observer/Metadata Sink virtual methods require ABI isolation from the older 1.x bindings.

set(LOGICRF_D3XX_ROOT "" CACHE PATH "Pinned external FTDI D3XX 1.1.8 input")
if(LOGICRF_D3XX_ROOT)
  file(SHA256 "${LOGICRF_D3XX_ROOT}/libftd3xx.so.1.1.8" LOGICRF_D3XX_SHA)
  if(NOT LOGICRF_D3XX_SHA STREQUAL "949445eb019a6fe3e314e54cc45af1b8ddb1fcdf0c377f1144a59143d0a08752")
    message(FATAL_ERROR "D3XX library pin mismatch")
  endif()
  foreach(LOGICRF_HEADER_PIN
      "ftd3xx.h:06ba4fc2d11680f974c1cfa76d504c61ef7325a089133e5247271d25d7111cb4"
      "Types.h:1f1669cea660bddffc9e42197201b2f17dec6c8a81fb2e12b268748d4ce9d66b"
      "libusb.h:e55b50eeedab44f0aa62ce6c2126d47ba80e8d9be1a67a62c27d2349bbf00c02")
    string(REPLACE ":" ";" LOGICRF_HEADER_PARTS "${LOGICRF_HEADER_PIN}")
    list(GET LOGICRF_HEADER_PARTS 0 LOGICRF_HEADER_NAME)
    list(GET LOGICRF_HEADER_PARTS 1 LOGICRF_HEADER_EXPECTED)
    file(SHA256 "${LOGICRF_D3XX_ROOT}/${LOGICRF_HEADER_NAME}" LOGICRF_HEADER_ACTUAL)
    if(NOT LOGICRF_HEADER_ACTUAL STREQUAL LOGICRF_HEADER_EXPECTED)
      message(FATAL_ERROR "D3XX/libusb header pin mismatch: ${LOGICRF_HEADER_NAME}")
    endif()
  endforeach()
  add_executable(logicrf-ft601-rx transport/ft601_rx.cc transport/h102_ft601_persistent_dma.c)

  target_include_directories(logicrf-ft601-duplex PRIVATE ${LOGICRF_D3XX_ROOT})
